
name: Build COD AI ANALYST EXE

on:
  workflow_dispatch:
  push:
    branches:
      - main

jobs:
  build:
    runs-on: windows-latest

    steps:
      - name: Download project
        uses: actions/checkout@v4

      - name: Install Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt
          pip install pyinstaller

      - name: Build EXE
        run: |
          pyinstaller --noconfirm --clean --onedir --name COD_AI_ANALYST run_app.py

      - name: Upload EXE folder
        uses: actions/upload-artifact@v4
        with:
          name: COD_AI_ANALYST-Windows
          path: dist/COD_AI_ANALYST/
